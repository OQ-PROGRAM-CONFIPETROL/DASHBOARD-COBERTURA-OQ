function getDashboardCoberturaOQ() { const ss = SpreadsheetApp.getActive(); // ========================================================== // NUEVA LÓGICA: CONTAR PERSONAL ÚNICO CAPACITADO // ========================================================== let tecnicosCapacitados = 0; const hojaBD = ss.getSheetByName("BASE DE DATOS"); if (hojaBD) { const dataBD = hojaBD.getDataRange().getValues(); // i = 1 para saltar la fila de los encabezados (Fila 1) for (let i = 1; i < dataBD.length; i++) { const nombre = String(dataBD[i][1] || "").trim(); // Columna B (Nombre) const cap1 = String(dataBD[i][4] || "").trim(); // Columna E (Cap 1) // Si la fila tiene un nombre válido y tiene texto en Cap 1 (evitando celdas vacías o guiones) if (nombre && cap1 && cap1 !== "" && cap1 !== "—" && cap1 !== "-") { tecnicosCapacitados++; } } } // ========================================================== const hoja = ss.getSheetByName("MATRIZ_COBERTURA_OQ"); if (!hoja) return { error: "No existe MATRIZ_COBERTURA_OQ" }; const data = hoja.getDataRange().getValues(); const tareas = []; let totalTareas = 0, cubiertas = 0, enProceso = 0, pendientes = 0; let totalTecnicos = 0, totalProveedores = 0; const prioridadStats = { "1": { total: 0, cubiertas: 0 }, "2": { total: 0, cubiertas: 0 }, "3": { total: 0, cubiertas: 0 } }; for (let i = 1; i < data.length; i++) { const row = data[i]; const tarea = String(row[1] || "").trim(); if (!tarea) continue; let requisito = (row[0] !== undefined && row[0] !== null) ? String(row[0]).trim() : ""; if (requisito.length === 10 && /^\d{3}\s-\s\d{4}$/.test(requisito)) { requisito = "0" + requisito; } else if (/^\d{3}$/.test(requisito)) { requisito = "0" + requisito; } if (!requisito) requisito = "—"; const clasificacion = String(row[2] || "").trim(); const prioridad = String(row[3] || "").trim(); const tecnicos = Number(row[4]) || 0; const proveedores = Number(row[5]) || 0; const cobertura = Number(row[6]) || 0; // Se mantiene tu mapeo actual impecable const estado = String(row[7] || "").trim().toUpperCase(); let fechaFinal = "—"; if (row[9] instanceof Date) { fechaFinal = Utilities.formatDate(row[9], Session.getScriptTimeZone(), "dd/MM/yyyy"); } else if (row[9]) { fechaFinal = String(row[9]).trim(); } const observacion = String(row[10] || "").trim(); totalTareas++; totalTecnicos += tecnicos; totalProveedores += proveedores; if (estado === "CUBIERTA") cubiertas++; else if (estado === "EN PROCESO") enProceso++; else pendientes++; if (prioridadStats[prioridad]) { prioridadStats[prioridad].total++; if (estado === "CUBIERTA") prioridadStats[prioridad].cubiertas++; } let nivel = "🔴"; if (cobertura >= 3) nivel = "🟢"; else if (cobertura >= 1) nivel = "🟡"; tareas.push({ requisito, tarea, clasificacion, prioridad, tecnicos, proveedores, cobertura, estado, fechaProgramada: fechaFinal, observacion, nivel }); } const prioridades = Object.keys(prioridadStats).map(pri => { const total = prioridadStats[pri].total; const cub = prioridadStats[pri].cubiertas; return { prioridad: pri, total, cubiertas: cub, porcentaje: total > 0 ? Math.round((cub / total) * 100) : 0 }; }); const porcentajeGeneral = totalTareas > 0 ? Math.round((cubiertas / totalTareas) * 100) : 0; const topCobertura = [...tareas].sort((a, b) => b.cobertura - a.cobertura).slice(0, 10); // Enviamos la nueva variable 'tecnicosCapacitados' al objeto de retorno return { totalTareas, cubiertas, enProceso, pendientes, porcentajeGeneral, totalTecnicos, totalProveedores, tecnicosCapacitados, prioridades, topCobertura, tareas }; }
